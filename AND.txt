cat > install_anydesk.sh <<'EOF'
#!/usr/bin/env bash

if [ "$EUID" -eq 0 ]; then
    SUDO=""
else
    SUDO="sudo"
fi

echo "==> Installing AnyDesk..."

if ! command -v wget >/dev/null 2>&1; then
    echo "Installing wget..."
    $SUDO apt-get update
    $SUDO apt-get install -y wget
fi

ARCH="$(dpkg --print-architecture)"

case "$ARCH" in
    amd64)
        URL="https://deb.anydesk.com/pool/main/a/anydesk/anydesk_8.0.4_amd64.deb"
        ;;
    arm64)
        URL="https://deb.anydesk.com/pool/main/a/anydesk/anydesk_8.0.4_arm64.deb"
        ;;
    armhf)
        URL="https://deb.anydesk.com/pool/main/a/anydesk/anydesk_6.3.0_armhf.deb"
        ;;
    i386)
        URL="https://deb.anydesk.com/pool/main/a/anydesk/anydesk_6.0.1_i386.deb"
        ;;
    *)
        echo "Unsupported architecture: $ARCH"
        exit 1
        ;;
esac

FILE="/tmp/$(basename "$URL")"

echo "Downloading $FILE..."
wget -c --tries=10 --timeout=30 -O "$FILE" "$URL" || exit 1

echo "Installing..."

$SUDO dpkg -i "$FILE" || {
    echo "Fixing dependencies..."
    $SUDO apt-get update
    $SUDO apt-get install -fy
    $SUDO dpkg -i "$FILE"
}

echo
echo "=================================="
echo " AnyDesk installed successfully!"
echo "=================================="
EOF

chmod +x install_anydesk.sh
./install_anydesk.sh
