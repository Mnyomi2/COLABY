
sudo sed -i 's/[a-z]*\.archive\.ubuntu\.com/archive.ubuntu.com/g' /etc/apt/sources.list

sudo apt-get -qq update >/dev/null 2>&1

sudo bash -c '
# ==============================================================================
# 💽 ULTIMATE LINUX ISO MANAGER & GRUB BOOTER (TURBO SPEED)
# ==============================================================================

TARGET_DIR="/live"
CUSTOM_GRUB="/etc/grub.d/40_custom"

# ⬇️ ADD NEW ISO URLS HERE IN THE FUTURE ⬇️
ISO_URLS=(
    "https://mirrors.cicku.me/linuxmint/iso/stable/22.3/linuxmint-22.3-cinnamon-64bit.iso"
    "https://releases.ubuntu.com/26.04/ubuntu-26.04-desktop-amd64.iso"
    "https://mirror.ueb.edu.ec/linuxliteos/isos/8.0/linux-lite-8.0-64bit.iso"
    "https://archive.org/download/windows-10-2022-07-11-19-29-38/Win11PE%20-%20x64.ISO"
)
# ⬆️ ==================================== ⬆️

# 1. Install aria2 for ultra-fast downloads if missing
if ! command -v aria2c >/dev/null 2>&1; then
    echo "⏳ Installing aria2 for Turbo Downloads..."
    DEBIAN_FRONTEND=noninteractive apt-get -qq update && DEBIAN_FRONTEND=noninteractive apt-get -qq -y install aria2
fi

mkdir -p "$TARGET_DIR"

# Make GRUB visible
sed -i "s/GRUB_TIMEOUT_STYLE=hidden/GRUB_TIMEOUT_STYLE=menu/g" /etc/default/grub
sed -i "s/GRUB_TIMEOUT=0/GRUB_TIMEOUT=10/g" /etc/default/grub

# Function to safely rebuild GRUB based ONLY on what is currently inside /live
rebuild_grub() {
    echo "⚙️  Rebuilding GRUB menu entries..."
    
    # Remove old auto-generated block to prevent duplicates
    sed -i "/# --- AUTO ISO ENTRIES START ---/,/# --- AUTO ISO ENTRIES END ---/d" "$CUSTOM_GRUB"
    
    echo "# --- AUTO ISO ENTRIES START ---" >> "$CUSTOM_GRUB"
    
    # Scan the folder and create a menu entry for every ISO found
    for iso_path in "$TARGET_DIR"/*.iso; do
        [ -e "$iso_path" ] || continue 
        filename=$(basename "$iso_path")
        
        # Ubuntu, Mint, and Lite all use the "casper" boot system
        cat <<EOF >> "$CUSTOM_GRUB"
menuentry "Live ISO: $filename" {
    set isofile="/live/$filename"
    search --no-floppy --file --set=root \$isofile
    loopback loop \$isofile
    linux (loop)/casper/vmlinuz boot=casper iso-scan/filename=\$isofile quiet splash ---
    initrd (loop)/casper/initrd
}
EOF
    done
    echo "# --- AUTO ISO ENTRIES END ---" >> "$CUSTOM_GRUB"
    
    echo "🔄 Updating Bootloader..."
    update-grub >/dev/null 2>&1
    echo "✅ GRUB Updated successfully!"
}

# Main Menu Loop
while true; do
    echo ""
    echo "=========================================="
    echo "      💽 ISO MANAGER MAIN MENU"
    echo "=========================================="
    echo "1) 🚀 Download & Add New ISO(s) (Aria2)"
    echo "2) 🗑️  Delete Existing ISO(s)"
    echo "3) ❌ Exit"
    echo "=========================================="
    read -p "Choose an option (1, 2, or 3): " main_choice

    case $main_choice in
        1)
            echo ""
            echo "--- AVAILABLE ISOs ---"
            for i in "${!ISO_URLS[@]}"; do
                filename=$(basename "${ISO_URLS[$i]}")
                echo "$((i+1))) $filename"
            done
            echo "----------------------"
            read -p "Enter numbers to download (e.g., 1, or 1 2 3): " dl_choices
            
            for c in $dl_choices; do
                idx=$((c-1))
                if [ -n "${ISO_URLS[$idx]}" ]; then
                    URL="${ISO_URLS[$idx]}"
                    FILENAME="$(basename "$URL")"
                    FILE_PATH="$TARGET_DIR/$FILENAME"
                    
                    echo "🚀 Downloading $FILENAME (Multi-thread Turbo)..."
                    
                    # Download using Aria2: 16 connections, 16 splits, 1MB min split size
                    aria2c -x 16 -s 16 -k 1M -d "$TARGET_DIR" -o "$FILENAME" "$URL"
                    
                    # Security check: Ensure it is a real file > 10MB (Catches 404 errors)
                    FILESIZE=$(stat -c%s "$FILE_PATH" 2>/dev/null || echo 0)
                    if [ "$FILESIZE" -lt 10000000 ]; then
                        echo "❌ ERROR: Download failed or URL does not exist. Deleting corrupted file."
                        rm -f "$FILE_PATH"
                        rm -f "${FILE_PATH}.aria2" # Clean up aria2 temp file if it exists
                    else
                        echo "✅ Download complete!"
                    fi
                fi
            done
            rebuild_grub
            ;;
            
        2)
            echo ""
            # Load existing ISOs into an array
            existing_isos=( "$TARGET_DIR"/*.iso )
            if [ ! -e "${existing_isos[0]}" ]; then
                echo "⚠️  No ISOs found in $TARGET_DIR to delete."
            else
                echo "--- INSTALLED ISOs ---"
                for i in "${!existing_isos[@]}"; do
                    echo "$((i+1))) $(basename "${existing_isos[$i]}")"
                done
                echo "----------------------"
                read -p "Enter numbers to DELETE (e.g., 1, or 1 2): " del_choices
                
                for c in $del_choices; do
                    idx=$((c-1))
                    if [ -n "${existing_isos[$idx]}" ]; then
                        echo "🗑️  Deleting $(basename "${existing_isos[$idx]}")...."
                        rm -f "${existing_isos[$idx]}"
                    fi
                done
                rebuild_grub
            fi
            ;;
            
        3)
            echo "👋 Exiting. Have a great day!"
            break
            ;;
            
        *)
            echo "❌ Invalid choice. Please press 1, 2, or 3."
            ;;
    esac
done
'
