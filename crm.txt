cat > install_chromeRM.sh <<'EOF'
#!/bin/bash
sudo sed -i 's/[a-z]*\.archive\.ubuntu\.com/archive.ubuntu.com/g' /etc/apt/sources.list



sudo DEBIAN_FRONTEND=noninteractive apt-get -qq update >/dev/null 2>&1 && command -v wget >/dev/null 2>&1 || { sudo DEBIAN_FRONTEND=noninteractive apt-get -qq update >/dev/null 2>&1 && sudo DEBIAN_FRONTEND=noninteractive apt-get -qq -y install wget gdebi-core >/dev/null 2>&1; }; wget -q https://dl.google.com/linux/direct/chrome-remote-desktop_current_amd64.deb -O /tmp/chrome-remote-desktop_current_amd64.deb && (sudo gdebi -n /tmp/chrome-remote-desktop_current_amd64.deb >/dev/null 2>&1 || sudo DEBIAN_FRONTEND=noninteractive apt-get -y install /tmp/chrome-remote-desktop_current_amd64.deb >/dev/null 2>&1 || { sudo DEBIAN_FRONTEND=noninteractive apt-get -fy install >/dev/null 2>&1 && sudo DEBIAN_FRONTEND=noninteractive apt-get -y install /tmp/chrome-remote-desktop_current_amd64.deb >/dev/null 2>&1; }) && rm -f /tmp/chrome-remote-desktop_current_amd64.deb && sudo systemctl enable chrome-remote-desktop@"$(whoami)".service >/dev/null 2>&1 && sudo systemctl start chrome-remote-desktop@"$(whoami)".service >/dev/null 2>&1 && sudo rm -rf /var/lib/apt/lists/*


sudo apt-get autoremove --purge -y && sudo apt-get autoclean -y && sudo apt-get clean -y && dpkg -l | awk '/^rc/ {print $2}' | xargs -r sudo dpkg --purge && sudo journalctl --vacuum-size=100M && sudo rm -rf /var/lib/apt/lists/* /tmp/* ~/.cache/* ~/.thumbnails ~/.local/share/Trash/* ~/.cache/thumbnails/* 2>/dev/null || true && echo "✅ System cleaned successfully."


DISPLAY= /opt/google/chrome-remote-desktop/start-host --code="4/0AXEQxIDp8KOF086lFfFpn6GDpeZesiNy4hMxULZ1kj-th9aySqVV-aA12xJIaisJ206CQg" --redirect-url="https://remotedesktop.google.com/_/oauthredirect" --name=$(hostname) --pin=000000

EOF

chmod +x install_chromeRM.sh
./install_chromeRM.sh
rm -f install_chromeRM.sh
